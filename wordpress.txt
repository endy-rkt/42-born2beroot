configure php and lighttpd
    $ sudo nano /etc/php/8.2/fpm/pool.d/www.conf
       ;listen = /run/php/php8.2-fpm.sock
        listen=127.0.0.1:9000

    $ sudo nano /etc/lighttpd/conf-available/15-fastcgi-php.conf and delete the first two to add these two
        "host" => "127.0.0.1",
        "port" => "9000",

    enable module to responsible for php to lighttpd communication
        $ sudo lighty-enable-mod fastcgi
        $ sudo lighty-enable-mod fastcgi-php
